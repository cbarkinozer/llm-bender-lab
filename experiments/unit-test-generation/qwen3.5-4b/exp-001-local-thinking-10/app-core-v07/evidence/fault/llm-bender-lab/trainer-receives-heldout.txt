import json
from pathlib import Path
FAILURES=[]
def pytest_configure(config):
    import experiment_workflow as m
    original=m.run_experiment
    def broken(root,name,config,rows,**kwargs):
        train=kwargs['trainer']
        kwargs['trainer']=lambda run,cfg,data:train(run,cfg,rows)
        return original(root,name,config,rows,**kwargs)
    m.run_experiment=broken

def pytest_runtest_logreport(report):
    if report.failed: FAILURES.append(dict(nodeid=report.nodeid,when=report.when,detail=report.longreprtext))
