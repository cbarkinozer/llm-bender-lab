def pytest_configure(config):
    from insignia.service import DashboardService
    from insignia.github import GitHubError
    async def broken(self):
        async with self.lock:
            try:
                candidate=await self.api.fetch('cbarkinozer',self.clock())
                self.snapshot=candidate
                self._save(candidate)

            else:self.error=None
            return self.read()
    DashboardService.refresh=broken
