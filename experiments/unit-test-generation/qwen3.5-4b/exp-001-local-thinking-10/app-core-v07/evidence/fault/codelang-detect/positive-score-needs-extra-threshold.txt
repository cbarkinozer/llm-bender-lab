import json
from pathlib import Path
FAILURES=[]
def pytest_configure(config):
    import codelang_detect.detector as m
    original=m.detect
    def broken(text):
        best=max((sum(w for pattern,w in rules if pattern.search(text)) for rules in m.PATTERNS.values()),default=0)
        return original(text) if best>=2 else 'unknown'
    m.detect=broken
    import codelang_detect as package
    package.detect=broken
    import codelang_detect.cli as cli
    cli.detect=broken

def pytest_runtest_logreport(report):
    if report.failed: FAILURES.append(dict(nodeid=report.nodeid,when=report.when,detail=report.longreprtext))
