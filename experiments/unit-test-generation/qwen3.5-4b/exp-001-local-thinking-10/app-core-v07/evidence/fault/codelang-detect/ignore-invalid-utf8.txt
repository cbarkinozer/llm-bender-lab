import json
from pathlib import Path
FAILURES=[]
def pytest_configure(config):
    import codelang_detect.cli as m
    OriginalFileType=m.argparse.FileType
    class LenientFileType(OriginalFileType):
        def __call__(self,name):
            stream=super().__call__(name)
            if hasattr(stream,'reconfigure'):stream.reconfigure(errors='ignore')
            return stream
    m.argparse.FileType=LenientFileType

def pytest_runtest_logreport(report):
    if report.failed: FAILURES.append(dict(nodeid=report.nodeid,when=report.when,detail=report.longreprtext))
