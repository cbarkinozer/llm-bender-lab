import json
from pathlib import Path
FAILURES=[]
def pytest_configure(config):
    import codelang_detect.detector as m
    def broken(text):
        if not text or not text.strip():return 'unknown'
        scores={label:sum(w for pattern,w in rules if pattern.search(text)) for label,rules in m.PATTERNS.items()}
        best=max(scores.values(),default=0)
        return sorted(label for label,score in scores.items() if score==best)[0] if best>0 else 'unknown'
    m.detect=broken
    import codelang_detect as package
    package.detect=broken
    import codelang_detect.cli as cli
    cli.detect=broken
