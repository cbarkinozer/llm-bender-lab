import json
from pathlib import Path
FAILURES=[]
def pytest_configure(config):
    import benchmark_workflow as m
    from pathlib import Path
    original=m.run_task
    def broken(task,**kwargs):
        result=original(task,**kwargs)
        root=Path(kwargs['root'])/task['id']
        (root/'generation.txt').write_text((root/'test_generated.py').read_text())
        return result
    m.run_task=broken

def pytest_runtest_logreport(report):
    if report.failed: FAILURES.append(dict(nodeid=report.nodeid,when=report.when,detail=report.longreprtext))
